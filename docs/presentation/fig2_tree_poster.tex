% Vertical (portrait) tree-search figure for the A0 poster.
% Same loop as REPORT_v6 Figure 2, re-proportioned top-to-bottom:
% the exit box moves to the BOTTOM of the spine so the whole column
% reads strictly downward and terminates at the deliverable.
\documentclass[tikz,border=2mm]{standalone}
\usetikzlibrary{positioning,arrows.meta,calc}
\begin{document}
\begin{tikzpicture}[
  font=\footnotesize,
  sbox/.style={draw=black!75, rounded corners=2pt, align=center, inner sep=5pt,
               fill=gray!8, text width=52mm},
  sidebox/.style={draw=black!75, rounded corners=2pt, align=center, inner sep=3.5pt,
                  fill=gray!3, text width=24mm, font=\scriptsize},
  injside/.style={sidebox, fill=blue!7, draw=blue!50},
  arr/.style={-{Stealth[length=2.6mm]}, thick},
  barr/.style={arr, blue!55},
  darr/.style={arr, dashed},
  lab/.style={font=\scriptsize, inner sep=1.5pt},
  node distance=6mm
]
% ----- main loop spine (top -> bottom) -----
\node[sbox] (phase) {\textbf{budget phase machine}\\exploit until every lineage plateaus\\$\to$ explore-burst: 5--8 fresh drafts\\$\to$ stop (cap 60 nodes; patience 20 after burst)};
\node[sbox, below=of phase] (select) {\textbf{select parent lineage}\\backtrack when a lineage plateaus\\(3 children without improvement)};
\node[sbox, below=of select] (propose) {\textbf{propose mutation}\\solo: seed / hyperparams /\\boundary push\\blend: members + weight search\\(first-class nodes over cached OOF)};
\node[sbox, below=of propose] (dedup) {\textbf{duplicate gate}\\SHA-256 config hash;\\a repeat burns budget,\\never re-evaluates};
\node[sbox, below=of dedup] (eval) {\textbf{evaluate in sandboxed child}\\CV score $\cdot$ SIGKILL on hang, node marked failed $\cdot$ blend cost guard};
\node[sbox, below=of eval] (add) {\textbf{add node, update global best}\\a solo breakthrough reopens\\plateaued blend lineages};
\foreach \a/\b in {phase/select, select/propose, propose/dedup, dedup/eval, eval/add}{\draw[arr] (\a) -- (\b);}
% ----- exit at the BOTTOM: the column terminates at the deliverable -----
\node[sbox, below=of add, fill=blue!8] (out)
  {\textbf{on stop: best node $\to$ submission}};
\draw[arr] (add) -- node[lab, right]{phase = stop} (out);
% ----- loop back, right side (re-enters at the select row) -----
\draw[arr] (add.east) -- ++(9mm,0) |- node[lab, pos=0.25, right, align=left]{next\\iteration} (select.east);
% ----- side inputs, left column -----
\node[injside, left=6mm of phase] (injcfg)
  {\textcolor{blue!60!black}{\textbf{model-level operators}}\\seeded as\\starting nodes};
\draw[barr] (injcfg) -- (phase);
\node[sidebox, densely dashed, left=6mm of propose] (prior)
  {priors from\\experience library};
\draw[darr] (prior) -- (propose);
\node[injside, left=6mm of eval] (injdata)
  {\textcolor{blue!60!black}{\textbf{data-level operators}}\\own arm each;\\arms race, winner\\takes full budget};
\draw[barr] (injdata) -- (eval);
\end{tikzpicture}
\end{document}
