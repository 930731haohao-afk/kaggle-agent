% ---------- 2 · problem ----------
\begin{frame}{The problem: agent benchmarks report noise}
  \begin{itemize}\itemsep=1.2ex
    \item LLM agents now solve Kaggle competitions end-to-end --- but comparisons are \alert{``one run, one score, higher wins''}
    \item Two questions current practice cannot answer:
      \begin{itemize}\itemsep=0.8ex
        \item \textbf{Positioning} --- where does our agent stand \emph{without} circular self-assessment?
        \item \textbf{Noise} --- score gaps sit at the 4th--5th decimal; are they real or sampling error?
      \end{itemize}
    \item This work: benchmark \textbf{three methodologically distinct agents} on \textbf{18 competitions}, with a significance test that separates signal from noise
  \end{itemize}
\end{frame}

% ---------- 4 · fairness ----------
\begin{frame}{A benchmark that cannot silently leak}
  \begin{itemize}\itemsep=1.2ex
    \item \textbf{18 competitions}: 13 Kaggle Playground + 5 older/niche ones; every submission scored on the \alert{real leaderboard} (late submission)
    \item Three agents share one machine $\Rightarrow$ mechanically enforced isolation:
      \begin{itemize}\itemsep=0.8ex
        \item data --- manifest-validated isolated roots
        \item compute --- mutex, one agent at a time
        \item instructions --- no recipes, no cross-agent hints; fix environments, never hint
      \end{itemize}
    \item Every number traceable to structured experiment records
  \end{itemize}
\end{frame}

% ---------- 5 · method ----------
\begin{frame}{Method: a paired test with a free error scale}
  \begin{itemize}\itemsep=1ex
    \item Kaggle scores each submission on \textbf{two disjoint test subsets} (public / private) --- two estimates of the same model; their gap = sampling noise
    \item Drift is \alert{shared}: s5e10 drifts $+0.00025/+0.00026/+0.00024$ --- common shift $0.00025$, mutual scatter $\mathbf{0.00002}$. Unpaired thresholds inflate noise $>$10$\times$
    \item Paired construction on the \textbf{between-agent gap} $d$ (common drift cancels):
  \end{itemize}
  \begin{align*}
    \operatorname{sign}(d_{\text{priv}}) &= \operatorname{sign}(d_{\text{pub}})
      && \text{order replicates on disjoint samples} \\
    |d_{\text{priv}}| &> |d_{\text{priv}} - d_{\text{pub}}|
      && \text{gap exceeds its own movement}
  \end{align*}
\end{frame}

% ---------- 7 · why ----------
\begin{frame}{Reading the runs: why agents win and fail}
  \begin{itemize}\itemsep=1.2ex
    \item \textbf{AIDE --- diagnosis cannot veto the score.} A bug fix silently switched to shuffled KFold on time-series data: metric jumped $18.4 \to 4.4$. Controlled re-run: \alert{4.5$\times$ optimism bias}. Nine steps spent tuning a leaked metric --- my-agent hit the same trap and quarantined it as ``DIAGNOSTIC''
    \item \textbf{NVIDIA --- votes $\neq$ quality.} An 89-vote workhorse kernel beat everyone; a 7{,}859-vote tutorial came last
    \item \textbf{Local CV $\neq$ leaderboard:} conclusions disagree in \alert{10 of 13} competitions (s3e19: CV says we win big, leaderboard says we lose)
  \end{itemize}
\end{frame}

% ---------- 8 · takeaways ----------
\begin{frame}{Takeaways}
  \begin{enumerate}\itemsep=1.5ex
    \item \textbf{Paired significance testing} for agent benchmarks --- error scale comes free from Kaggle's public/private split; 26\% of ``verdicts'' are noise without it
    \item \textbf{CV-only benchmarks are unreliable} --- submit to the real leaderboard (10/13 disagree)
    \item \textbf{State the evaluation-set composition} --- rankings are properties of method $\times$ competition distribution, not of the method alone
  \end{enumerate}
